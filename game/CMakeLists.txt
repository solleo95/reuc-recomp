  add_library(uc_game STATIC ${UC_GENERATED_SOURCES})

  if(UC_TRACE)
    target_compile_definitions(uc_game PUBLIC UC_TRACE=1)
  endif()

  target_link_libraries(uc PRIVATE uc_game uc_common uc_platform uc_options uc_vulkan)

    COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL3-shared> $<TARGET_FILE_DIR:uc> VERBATIM)
