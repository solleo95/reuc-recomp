# Compiler cache: the 108 generated game files take about 25 CPU minutes, and a revert, a switch between two states or a clean tree finds them in the cache.
# Used when ccache is installed (winget install Ccache.Ccache); -DUC_CCACHE=OFF turns it off. MSVC-style /Zi debug info (a shared PDB) cannot be cached: the build then compiles as before.
option(UC_CCACHE "use ccache as the compiler launcher when it is installed" ON)
if(UC_CCACHE)
  file(GLOB uc_ccache_hints "$ENV{LOCALAPPDATA}/Microsoft/WinGet/Packages/Ccache.Ccache_*/ccache-*")   # where winget unpacks it: a shell started before the install does not have it on PATH
  find_program(UC_CCACHE_PROGRAM ccache HINTS ${uc_ccache_hints} "$ENV{LOCALAPPDATA}/Microsoft/WinGet/Links")
  if(UC_CCACHE_PROGRAM)
    # pch_defines,time_macros: lets ccache use the precompiled headers of the game/runtime targets (without it it reports "could not use precompiled header" and compiles those files uncached).
    # The settings go with the call, so a shell that does not have the user's ccache configuration (PowerShell and Git Bash read different ones) behaves the same.
    set(CMAKE_C_COMPILER_LAUNCHER "${CMAKE_COMMAND}" -E env CCACHE_SLOPPINESS=pch_defines,time_macros "${UC_CCACHE_PROGRAM}")
    set(CMAKE_CXX_COMPILER_LAUNCHER "${CMAKE_COMMAND}" -E env CCACHE_SLOPPINESS=pch_defines,time_macros "${UC_CCACHE_PROGRAM}")
    message(STATUS "ccache: ${UC_CCACHE_PROGRAM}")
  endif()
endif()
# CPU level of the Windows build. The release holds two game programs: uc_avx2.exe (x86-64-v3: AVX2, FMA, BMI) and uc_compat.exe (x86-64-v2: SSE4.2, Nehalem and newer); uc.exe
# (game/cpu_dispatch.cpp) starts the one this CPU runs. build.bat builds the second one in build-compat (-DUC_ARCH=x86-64-v2) together with the launcher and the updater,
# which have to start on every CPU. -ffp-contract=off: only the v3 build could fuse a*b+c into an FMA, and the two builds must compute the same floats (netplay hashes the state).
set(UC_ARCH "x86-64-v3" CACHE STRING "CPU level the Windows code is compiled for (clang -march)")
set(UC_COMPAT_DIR "${CMAKE_SOURCE_DIR}/build-compat" CACHE PATH "build folder of the x86-64-v2 programs that the release install takes (see UC_ARCH)")
if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
  add_compile_options(/clang:-march=${UC_ARCH} /clang:-ffp-contract=off)
endif()
if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
  add_compile_options(-Wno-deprecated-declarations)   # "unsafe CRT function" (getenv, fopen), winsock and std::filesystem::u8path notes in the tools and third-party code are noise
endif()
# Guest memory access tracer (runtime/mem_trace.h): records the loads and stores of selected translated functions to find work that can run in parallel. Diagnostics only,
# 20x slower; build it in its own folder (the leaf functions then run as translated code, so every access is seen).
option(UC_TRACE_MEM "record guest memory accesses of the functions in UC_TRACE_MEM_FUNCS" OFF)
if(UC_TRACE_MEM)
  add_compile_definitions(UC_TRACE_MEM=1)
endif()
# Profile-guided optimization: ~8% more frames in gameplay. `python scripts/pgo.py` records project/pgo/uc.profdata from a harness run of an
# instrumented build; while that file exists every build uses it (stale parts are ignored silently, re-record after big changes).
option(UC_PGO_GENERATE "build an instrumented uc.exe that records a profile (scripts/pgo.py)" OFF)
set(UC_PGO_PROFILE "${CMAKE_SOURCE_DIR}/project/pgo/uc.profdata" CACHE FILEPATH "profile for the optimizer; used when the file exists")
if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
  if(UC_PGO_GENERATE)
    add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/clang:-fprofile-instr-generate>)
    get_filename_component(uc_llvm_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
    file(GLOB uc_profile_runtime "${uc_llvm_bin}/../lib/clang/*/lib/windows/clang_rt.profile-x86_64.lib")
    if(NOT uc_profile_runtime)
      message(FATAL_ERROR "clang_rt.profile-x86_64.lib not found next to ${CMAKE_CXX_COMPILER}")
    endif()
    add_link_options("${uc_profile_runtime}")
  elseif(EXISTS "${UC_PGO_PROFILE}")
    # A new profile must rebuild every object; mixed ThinLTO profile summaries cannot link.
    file(SHA256 "${UC_PGO_PROFILE}" uc_pgo_hash)
    set(uc_pgo_input "${CMAKE_BINARY_DIR}/pgo/${uc_pgo_hash}.profdata")
    configure_file("${UC_PGO_PROFILE}" "${uc_pgo_input}" COPYONLY)
    add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/clang:-fprofile-instr-use=${uc_pgo_input}> -Wno-profile-instr-unprofiled -Wno-profile-instr-missing -Wno-profile-instr-out-of-date)
    message(STATUS "PGO profile: ${UC_PGO_PROFILE}")
  endif()
endif()
# ThinLTO across the translated game, runtime and renderer (lets the small native helpers be inlined into generated code); the link is slow.
option(UC_THINLTO "build with ThinLTO (build.bat and scripts/dev.bat turn it on)" OFF)
if(UC_THINLTO AND WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
  add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/clang:-flto=thin>)
  add_link_options("/lldltocache:${CMAKE_BINARY_DIR}/lto-cache")
endif()
if(UC_LINUX AND CMAKE_CXX_COMPILER_ID STREQUAL "Clang")   # the Steam Deck build: scripts/linux/pgo.py records project/pgo/uc-linux.profdata (names differ from the Windows profile)
  set(UC_PGO_LINUX_PROFILE "${CMAKE_SOURCE_DIR}/project/pgo/uc-linux.profdata" CACHE FILEPATH "profile for the Linux optimizer; used when the file exists")
  if(UC_PGO_GENERATE)
    get_filename_component(uc_llvm_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
    file(GLOB uc_profile_include "${uc_llvm_bin}/../lib/clang/*/include")
    add_library(uc_profile_rt STATIC runtime/linux_profile_rt.c)   # the sysroot's runtime is LLVM 11's: this one writes the compiler's own raw format (see the file)
    target_include_directories(uc_profile_rt PRIVATE ${uc_profile_include})
    add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-fprofile-instr-generate> "SHELL:$<$<COMPILE_LANGUAGE:CXX>:-mllvm -enable-value-profiling=false>")
    # linked by the game only (game/CMakeLists.txt): a global link_libraries would reach third-party targets with export sets
  elseif(EXISTS "${UC_PGO_LINUX_PROFILE}")
    file(SHA256 "${UC_PGO_LINUX_PROFILE}" uc_pgo_hash)
    set(uc_pgo_input "${CMAKE_BINARY_DIR}/pgo/${uc_pgo_hash}.profdata")
    configure_file("${UC_PGO_LINUX_PROFILE}" "${uc_pgo_input}" COPYONLY)
    add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-fprofile-instr-use=${uc_pgo_input}> -Wno-profile-instr-unprofiled -Wno-profile-instr-missing -Wno-profile-instr-out-of-date)
    message(STATUS "PGO profile: ${UC_PGO_LINUX_PROFILE}")
  endif()
endif()
if(UC_THINLTO AND UC_LINUX)   # the Linux (Steam Deck) build: scripts/linux/build.ps1 -ThinLTO (own build folder)
  add_compile_options(-flto=thin)
  add_link_options(-flto=thin "-Wl,--thinlto-cache-dir=${CMAKE_BINARY_DIR}/lto-cache")
endif()

# Every test runs on a desktop of its own (tools/hidden_run): the windows the tests open never appear over what the user is doing and never take the focus, however ctest is
# started. Set before any test program is created (CMake copies it into each program's TEST_LAUNCHER). UC_TESTS_VISIBLE=1 shows them.
add_subdirectory(tools/hidden_run)
set(CMAKE_TEST_LAUNCHER "$<TARGET_FILE:uc_hidden_run>")   # the tests that are programs of this build
# The tests that are a script (python, cmake -P) can start programs with windows too (test_update_end_to_end runs the launcher and the updater): add_test puts the launcher
# in front of their command as well. (The original command stays reachable as _add_test; a program of this build already has the launcher.)
function(add_test)
  if(NOT ARGV0 STREQUAL "NAME")
    _add_test(${ARGV})
    return()
  endif()
  set(arguments ${ARGN})
  list(FIND arguments "COMMAND" at)
  if(at GREATER_EQUAL 0)
    math(EXPR first "${at}+1")
    list(GET arguments ${first} program)
    if(NOT TARGET "${program}")
      list(INSERT arguments ${first} "$<TARGET_FILE:uc_hidden_run>")
    endif()
  endif()
  _add_test(${arguments})
endfunction()
